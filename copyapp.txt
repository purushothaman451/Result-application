#captcha generation
image_captcha = ImageCaptcha(width=200,height=90)
def generate_random_string(length=6):
    """Generates a random alphanumeric string."""
    letters = string.ascii_letters + string.digits
    return ''.join(random.choice(letters) for i in range(length))

@app.route('/captcha-image', methods=['GET'])
def get_captcha():
    captcha_text = generate_random_string(6)
    session['captcha'] = captcha_text.upper() # Store uppercase for case-insensitive check
    data = image_captcha.generate(captcha_text)
    
    return send_file(
        io.BytesIO(data.read()),
        mimetype='image/png',
        download_name='captcha.png'
    )

@app.route('/resultverify/input', methods=['GET','POST'])
def verify_login():
    """Handles the standard user/student login from the frontend with CAPTCHA."""
    
    username = request.form.get('reg_id', '').strip()
    password = request.form.get('dob', '').strip()
    user_captcha_input = request.form.get('input', '').strip()

    # 1. CAPTCHA Verification
    stored_captcha = session.get('captcha')
    session.pop('captcha', None) # Clear immediately to prevent reuse

    if not stored_captcha:
        flash("CAPTCHA expired. Please refresh the page.")
        return redirect(url_for('result'))
    
    if user_captcha_input.upper() != stored_captcha:
        flash("Invalid CAPTCHA. Try again.")
        return redirect(url_for('result'))

    # 2. Database Verification (Using SQLAlchemy)
    user = User.query.filter_by(username=username).first()
    
    if check_password_hash(user.username,) and check_password_hash(user.password, password):
        # SUCCESS! Log them in
        flash("Invalid userid or password")
        login_user(user)
        return redirect(url_for('result'))
    else:
        flash("Invalid Register Number or Date of Birth.")
        return redirect(url_for('result'))


@app.route('/admin', methods=['GET', 'POST'])

@app.route('/admin/import_users', methods=['GET', 'POST'])
@login_required  # Secure this so only logged-in admins can upload files
def import_users():
    if request.method == 'POST':
        # 1. Check if a file was actually uploaded
        if 'file' not in request.files:
            flash('No file part')
            return redirect(request.url)
            
        file = request.files['file']
        
        if file.filename == '':
            flash('No selected file')
            return redirect(request.url)

        if file:
            try:
                # 2. Read the file using Pandas (handles both CSV and Excel)
                if file.filename.endswith('.csv'):
                    df = pd.read_csv(file)
                elif file.filename.endswith(('.xls', '.xlsx')):
                    df = pd.read_excel(file)
                else:
                    flash('Invalid file format. Please upload a CSV or Excel file.')
                    return redirect(request.url)

                df.columns = df.columns.str.strip()
                # 3. Loop through each row of the file and add to the database
                for index, row in df.iterrows():
                    username1 = str(row['reg_id']).strip()
                    plain_password = str(row['dob']).strip()
                    student_name = str(row['name']).strip()
                    student_sem = str(row['semester']).strip()
                    s_year = str(row['year']).strip()
                    s_institute = str(row['institute']).strip()

                    
                    # Optional: Grab student info if you have columns for them
                    # location = str(row['location']).strip() if 'location' in row else None

                    # Check if the user already exists to avoid duplicates
                    existing_user = student_user.query.filter_by(reg_id=username1).first()
                    if not existing_user:
                        # Hash the password securely before saving
                        hashed_username1 = generate_password_hash(username1, method='pbkdf2:sha256')
                        hashed_dob = generate_password_hash(plain_password, method='pbkdf2:sha256')
                        hashed_student_name = generate_password_hash(student_name, method='pbkdf2:sha256')
                        hashed_student_sem = generate_password_hash(student_sem, method='pbkdf2:sha256')
                        hashed_s_year = generate_password_hash(s_year,method='pbkdf2:sha256')
                        hashed_s_institute = generate_password_hash(s_institute, method='pbkdf2:sha256')
                        
                        # Create new user (The ID is auto-assigned by the database)
                        new_user = student_user(reg_id=hashed_username1, dob=hashed_dob, name=hashed_student_name, semester=hashed_student_sem, year=hashed_s_year, institute=hashed_s_institute)
                        db.session.add(new_user)

                # Save everything to the database at once
                db.session.commit()
                flash('File imported and users added successfully!')
                return redirect(url_for('admin_dashboard'))

            except Exception as e:
                db.session.rollback()  # Undo changes if an error happens
                flash(f'An error occurred during import: {str(e)}')
                return redirect(request.url)

    return render_template('import.html')

                        )
                        db.session.add(new_result)
                    else:
                        # Optional: Print or log that a result was skipped because the student doesn't exist
